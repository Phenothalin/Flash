# ==============================================================================
# RR Module: Rachford-Rice Flash Solver
# ==============================================================================

# ------------------------------------------------------------------------------
# Library Target
# ------------------------------------------------------------------------------

# Collect source files
file(GLOB RR_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
file(GLOB RR_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp"
                     "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h")

# Create static library
add_library(rrflash STATIC ${RR_SOURCES} ${RR_HEADERS})

target_include_directories(rrflash PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/../Thermo/include
    ${CMAKE_CURRENT_SOURCE_DIR}/../PhaseStability/include
)

target_link_libraries(rrflash PUBLIC
    thermo
    phase_stability
    Eigen3::Eigen
)

# ------------------------------------------------------------------------------
# Test Executables
# ------------------------------------------------------------------------------

# Helper function to create test executables
function(add_rr_test TEST_NAME SOURCE_FILE)
    add_executable(${TEST_NAME} ${SOURCE_FILE})

    target_include_directories(${TEST_NAME} PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/include
        ${CMAKE_CURRENT_SOURCE_DIR}/../Thermo/include
        ${CMAKE_CURRENT_SOURCE_DIR}/../PhaseStability/include
    )

    target_link_libraries(${TEST_NAME} PRIVATE
        rrflash
        GTest::gtest
        GTest::gtest_main
    )

    set_thermopack_rpath(${TEST_NAME})
endfunction()

# Two-phase RR flash test
add_rr_test(RRflash_test
    tests/RRflash_test.cpp
)

# Three-phase VLL test
add_rr_test(RR_vll_test
    tests/RR_vll_test.cpp
)
