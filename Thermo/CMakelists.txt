# Thermo 层：封装 thermopack，提供 C++ backend

# 收集源码和头文件（你可以改成手动列出）
file(GLOB THERMO_SOURCES
     "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
)
file(GLOB THERMO_HEADERS
     "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp"
     "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h"
)

if(THERMO_SOURCES)
  # 有 .cpp 的情况：正常生成静态库
  add_library(thermo STATIC
      ${THERMO_SOURCES}
      ${THERMO_HEADERS}
  )

  target_include_directories(thermo
      PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
  )

  target_link_libraries(thermo
      PUBLIC
        thermopack
        Eigen3::Eigen
  )
else()
  # 暂时没有源码：做成 header-only 的 INTERFACE 库
  add_library(thermo INTERFACE)

  target_include_directories(thermo
      INTERFACE
        ${CMAKE_CURRENT_SOURCE_DIR}/include
  )

  target_link_libraries(thermo
      INTERFACE
        thermopack
        Eigen3::Eigen
  )
endif()
