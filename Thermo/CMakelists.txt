# ==============================================================================
# Thermo Module: ThermoPack C++ Backend Wrapper
# ==============================================================================

# ------------------------------------------------------------------------------
# Library Target
# ------------------------------------------------------------------------------

# Collect source files
file(GLOB THERMO_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
file(GLOB THERMO_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp"
                         "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h")

# Create library (STATIC or INTERFACE depending on source availability)
if(THERMO_SOURCES)
    # Static library with implementation
    add_library(thermo STATIC ${THERMO_SOURCES} ${THERMO_HEADERS})

    target_include_directories(thermo PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
    )

    target_link_libraries(thermo PUBLIC
        thermopack
        Eigen3::Eigen
    )
else()
    # Header-only interface library
    add_library(thermo INTERFACE)

    target_include_directories(thermo INTERFACE
        ${CMAKE_CURRENT_SOURCE_DIR}/include
    )

    target_link_libraries(thermo INTERFACE
        thermopack
        Eigen3::Eigen
    )
endif()

# ------------------------------------------------------------------------------
# Test Executables
# ------------------------------------------------------------------------------

# Helper function to create test executables
function(add_thermo_test TEST_NAME SOURCE_FILE)
    add_executable(${TEST_NAME} ${SOURCE_FILE})

    target_include_directories(${TEST_NAME} PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/include
    )

    target_link_libraries(${TEST_NAME} PRIVATE
        thermopack
        Eigen3::Eigen
    )

    set_thermopack_rpath(${TEST_NAME})
endfunction()

# Compressibility factor / Basic EOS test
add_thermo_test(compressibility_factor_test
    tests/compressibility_factor_test.cpp
)

# ThermoPack sanity check (optional)
if(BUILD_TP_SANITY_TEST)
    add_thermo_test(tp_thermo_check
        tests/tp_thermo_check.cpp
    )
endif()
