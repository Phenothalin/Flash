# Helper function to create test executables
# Usage: add_rand_test(test_name source_file [USE_GTEST] [EXTRA_INCLUDES dir1 dir2...])
function(add_rand_test TEST_NAME SOURCE_FILE)
    # Parse optional arguments
    set(options USE_GTEST)
    set(multiValueArgs EXTRA_INCLUDES)
    cmake_parse_arguments(ARG "${options}" "" "${multiValueArgs}" ${ARGN})

    add_executable(${TEST_NAME} ${SOURCE_FILE})

    # Base include directories
    set(INCLUDE_DIRS
        ${CMAKE_CURRENT_SOURCE_DIR}/include
        ${CMAKE_CURRENT_SOURCE_DIR}/../Thermo/include
    )

    # Add extra includes if specified
    if(ARG_EXTRA_INCLUDES)
        list(APPEND INCLUDE_DIRS ${ARG_EXTRA_INCLUDES})
    endif()

    target_include_directories(${TEST_NAME} PRIVATE ${INCLUDE_DIRS})

    # Base libraries
    set(LINK_LIBS randflash)

    # Add GTest if requested
    if(ARG_USE_GTEST)
        list(APPEND LINK_LIBS GTest::gtest GTest::gtest_main)
    endif()

    target_link_libraries(${TEST_NAME} PRIVATE ${LINK_LIBS})

    # Set RPATH for dynamic library loading
    set_thermopack_rpath(${TEST_NAME})
endfunction()
