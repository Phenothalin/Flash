# RPATH helper function for thermopack
function(set_thermopack_rpath target_name)
  if(NOT THERMOPACK_USE_STATIC)
    if(APPLE)
      get_target_property(_TP_LIB thermopack IMPORTED_LOCATION)
      get_filename_component(_TP_LIBDIR "${_TP_LIB}" DIRECTORY)
      set_target_properties(${target_name} PROPERTIES
        BUILD_WITH_INSTALL_RPATH ON
        BUILD_RPATH "${_TP_LIBDIR}"
        INSTALL_RPATH "${_TP_LIBDIR};@loader_path"
      )
    elseif(WIN32)
      get_target_property(_TP_LIB thermopack IMPORTED_LOCATION)
      add_custom_command(TARGET ${target_name} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${_TP_LIB}" $<TARGET_FILE_DIR:${target_name}>
      )
    endif()
  endif()
endfunction()
