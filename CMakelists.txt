cmake_minimum_required(VERSION 3.18)

# ===================== vcpkg 工具链 & triplet =====================
if(NOT DEFINED ENV{VCPKG_ROOT})
  message(FATAL_ERROR "VCPKG_ROOT not set. Please export VCPKG_ROOT first.")
endif()

set(CMAKE_TOOLCHAIN_FILE
    "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake"
    CACHE STRING "Vcpkg toolchain file"
)

if(NOT DEFINED VCPKG_TARGET_TRIPLET)
  if(APPLE)
    set(VCPKG_TARGET_TRIPLET "arm64-osx" CACHE STRING "vcpkg triplet" FORCE)
  elseif(WIN32)
    set(VCPKG_TARGET_TRIPLET "x64-windows" CACHE STRING "vcpkg triplet" FORCE)
  endif()
endif()

project(Flash LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ===================== ThermoPack =====================
if(DEFINED ENV{THERMOPACK_DIR})
  list(PREPEND CMAKE_PREFIX_PATH "$ENV{THERMOPACK_DIR}")
endif()

find_package(THERMOPACK CONFIG REQUIRED)

find_package(GTest REQUIRED)
# ===================== Eigen（通过 vcpkg） =====================
find_package(Eigen3 CONFIG REQUIRED)

# 为了保险，把 vcpkg 安装路径补进 CMAKE_PREFIX_PATH
list(APPEND CMAKE_PREFIX_PATH
     "$ENV{VCPKG_ROOT}/installed/${VCPKG_TARGET_TRIPLET}"
)

# ===================== 选项：是否编译 ThermoPack 检查程序 =====================
option(BUILD_TP_SANITY_TEST "Build ThermoPack sanity check test" ON)

# ===================== 子目录 =====================
add_subdirectory(Thermo)
# add_subdirectory(RR)
add_subdirectory(RAND)
